  16 | class AIOrchestrator:
  17 |     """
  18 |     Coordinates the execution pipeline of AI models for the LIMATA service.
  19 |     
  20 |     The AIOrchestrator serves as the central conductor for inference requests. 
  21 |     It focuses strictly on the execution flow—receiving inputs, routing them to 
  22 |     the correct models via the ModelLoader, and returning the outputs. It is 
  23 |     completely decoupled from model initialization, lifecycle management, and metadata.
  24 |     """
  25 | 
  26 |     def __init__(self, loader: ModelLoader) -> None:
  27 |         """
  28 |         Initializes the AIOrchestrator with a required ModelLoader dependency.
  29 |         
  30 |         Args:
  31 |             loader (ModelLoader): The loader service responsible for supplying 
  32 |                                   fully initialized AI models on demand.
  33 |         """
  34 |         self._loader: ModelLoader = loader
  35 |         self._spatial_engine: SpatialAnalysisEngine = SpatialAnalysisEngine()
  36 |         self._placement_engine: PlacementEvaluationEngine = PlacementEvaluationEngine()
  37 | 
  38 |     def analyze_image(self, image: Any) -> DetectionResult:
  39 |         """
  40 |         Executes object detection on the provided image using a YOLO model.
  41 |         
  42 |         This method retrieves the appropriate YOLO model from the loader, 
  43 |         passes the input image for inference, and cleanly converts the raw 
  44 |         third-party tensors into a LIMATA-native DetectionResult.
  45 |         
  46 |         Args:
  47 |             image (Any): The input image to be analyzed.
  48 |                          
  49 |         Returns:
  50 |             DetectionResult: A decoupled, framework-agnostic result containing 
  51 |                              bounding boxes, confidences, and labels.
  52 |                              
  53 |         Raises:
  54 |             AIInferenceException: If inference fails during execution.
  55 |         """
  56 |         try:
  57 |             start_time = time.perf_counter()
  58 |             
  59 |             raw_results = self._detect_objects(image, ModelNames.YOLO)
  60 |             
  61 |             end_time = time.perf_counter()
  62 |             inference_time_ms = (end_time - start_time) * 1000.0
  63 |             
  64 |             return self._convert_results(raw_results, ModelNames.YOLO, inference_time_ms)
  65 |             
  66 |         except Exception as e:
  67 |             raise AIInferenceException(f"Failed to analyze image: {str(e)}") from e

  71 |         Executes depth estimation on the provided image using a MiDaS model.

  73 |         Args:

  96 |     def analyze_spatial_layout(self, image: Any) -> SpatialAnalysisResult:
  97 |         """
  98 |         Executes both object detection and depth estimation pipelines, then

 101 |         Args:

 123 |     def evaluate_placement(self, image: Any, furniture: FurnitureMetadata) -> PlacementEvaluationResult:
 124 |         """
 125 |         Convenience method that runs the entire spatial layout pipeline and then evaluates
 126 |         whether the provided furniture constraints are suitable for the room.
 127 |         
 128 |         Args:
 129 |             image (Any): The input image to analyze.
 130 |             furniture (FurnitureMetadata): Constraints of the furniture item.
 131 |             
 132 |         Returns:
 133 |             PlacementEvaluationResult: The deterministic evaluation output.
 134 |         """
 135 |         try:
 136 |             # 1. Analyze spatial layout (perceive the room)
 137 |             spatial_result = self.analyze_spatial_layout(image)
 138 |             
 139 |             # 2. Heuristically evaluate placement against the spatial layout
 140 |             # Here we need image height and width to get total area.
 141 |             # DetectionResult stores it, but we can also retrieve it directly from the image shape.
 142 |             # Since analyze_spatial_layout abstracts the original image, we'll extract shape here.
 143 |             # Assuming OpenCV format image (H, W, C)
 144 |             try:
 145 |                 import numpy as np
 146 |                 if isinstance(image, np.ndarray):
 147 |                     image_height, image_width = image.shape[:2]
 148 |                 else:
 149 |                     image_height, image_width = 480, 640 # fallback
 150 |             except:
 151 |                 image_height, image_width = 480, 640 # fallback
 152 |                 
 153 |             return self._placement_engine.evaluate(spatial_result, furniture, image_width, image_height)

 155 |             raise AIInferenceException(f"Failed to evaluate placement: {str(e)}") from e

 159 |         Private helper to request the model and execute inference.
 160 |         """

 162 |         return model(image)

 164 |     def _convert_results(self, raw_results: Any, model_name: str, inference_time_ms: float) -> DetectionResult:
 165 |         """
 166 |         Private helper to delegate raw results to the converter.
 167 |         """
 168 |         return convert_yolo_results(
 169 |             raw_results, 
 170 |             model_name=model_name, 
 171 |             inference_time_ms=inference_time_ms
 172 |         )

 174 |     def _estimate_depth(self, image: Any, model_name: str) -> Any:
 175 |         """
 176 |         Private helper to request the MiDaS model and execute inference.
 177 |         """
 178 |         import torch
 179 |         model = self._loader.get_model(model_name)
 180 |         
 181 |         # Determine device (CPU for now as safe default)
 182 |         device = torch.device("cpu")
 183 |         model.to(device)
 184 |         
 185 |         # Apply transform
 186 |         input_batch = model.transform(image).to(device)
 187 |         
 188 |         with torch.no_grad():
 189 |             prediction = model(input_batch)
 190 |             
 191 |             # Interpolate to original size
 192 |             # image is expected to be a numpy array of shape (H, W, C)
 193 |             height, width = image.shape[:2]
 194 |             prediction = torch.nn.functional.interpolate(
 195 |                 prediction.unsqueeze(1),
 196 |                 size=(height, width),
 197 |                 mode="bicubic",
 198 |                 align_corners=False,
 199 |             ).squeeze()
 200 |             
 201 |         # We return the tensor, which converters.py will wrap in a DepthResult
 202 |         return (prediction, width, height)
 203 |

 205 |         """
 206 |         Private helper to delegate raw depth results to the converter.
 207 |         """
 208 |         from .converters import convert_midas_results
 209 |         return convert_midas_results(
 210 |             raw_results, 
 211 |             model_name=model_name, 
 212 |             inference_time_ms=inference_time_ms
 213 |         )

 100 |     def load_model(self, model_name: str) -> Any:
 101 |         """
 102 |         Loads a YOLO model into memory, adhering to thread-safe lazy loading principles.
 103 |         
 104 |         Args:
 105 |             model_name (str): The unique identifier of the model to load.
 106 |             
 107 |         Returns:
 108 |             Any: The loaded YOLO model instance.
 109 |             
 110 |         Raises:
 111 |             ModelNotFoundException: If the model is not registered in the ModelRegistry.
 112 |             Exception: Any error encountered during model loading.
 113 |         """
 114 |         # 1. Verify the model exists in ModelRegistry.
 115 |         if not self._registry.exists(model_name):
 116 |             raise ModelNotFoundException(model_name)
 117 | 
 118 |         # 2. If already loaded, immediately return the cached instance.
 119 |         if self.is_loaded(model_name):
 120 |             return self._loaded_models[model_name]
 121 | 
 122 |         # Ensure lock and status exist for dynamically registered models
 123 |         if model_name not in self._loading_locks:
 124 |             self._loading_locks[model_name] = Lock()
 125 |             self._runtime_status[model_name] = RuntimeStatus(state=ModelState.NOT_LOADED)
 126 | 
 127 |         # 3. Acquire the model-specific loading lock.
 128 |         with self._loading_locks[model_name]:
 129 |             
 130 |             # 4. Double-check whether another thread already loaded the model.
 131 |             if self.is_loaded(model_name):
 132 |                 return self._loaded_models[model_name]
 133 | 
 134 |             # 5. Update RuntimeStatus to LOADING.
 135 |             self._runtime_status[model_name].state = ModelState.LOADING
 136 |             self._runtime_status[model_name].last_updated = datetime.now(timezone.utc)
 137 |             self._runtime_status[model_name].error_message = None
 138 | 
 139 |             try:
 140 |                 # 6. Read the metadata from ModelRegistry.
 141 |                 metadata = self._registry.get(model_name)
 142 | 
 143 |                 # 7. Load the model
 144 |                 if model_name == "midas":
 145 |                     import torch
 146 |                     # Download/Load MiDaS and its transforms
 147 |                     torch.hub.set_dir('models/midas')
 148 |                     model = torch.hub.load("intel-isl/MiDaS:master", "MiDaS_small", trust_repo=True)
 149 |                     midas_transforms = torch.hub.load("intel-isl/MiDaS:master", "transforms", trust_repo=True)
 150 |                     # Attach the transform to the model for convenience in the orchestrator
 151 |                     model.transform = midas_transforms.small_transform
 152 |                     # Evaluate mode
 153 |                     model.eval()
 154 |                 else:
 155 |                     # Default to YOLO (original architecture assumption)
 156 |                     model = YOLO(metadata.weights_path)
 157 | 
 158 |                 # 8. Store the model inside _loaded_models.
 159 |                 self._loaded_models[model_name] = model
 160 | 
 161 |                 # 9. Update RuntimeStatus to READY.
 162 |                 self._runtime_status[model_name].state = ModelState.READY
 163 |                 self._runtime_status[model_name].last_updated = datetime.now(timezone.utc)
 164 |                 
 165 |                 # 10. Return the loaded model.
 166 |                 return model
 167 | 
 168 |             except Exception as e:
 169 |                 # Error Handling: Update to FAILED, store message, and re-raise.
 170 |                 self._runtime_status[model_name].state = ModelState.FAILED
 171 |                 self._runtime_status[model_name].last_updated = datetime.now(timezone.utc)
 172 |                 self._runtime_status[model_name].error_message = str(e)
 173 |                 raise ModelLoadException(f"Failed to load model {model_name}: {e}") from e
 174 |

 179 |         This is the primary public entry point for accessing AI models. It guarantees

 183 |             model_name (str): The unique identifier of the requested model.

   5 | def calculate_congestion_index(spatial_result: SpatialAnalysisResult, image_width: int, image_height: int) -> float:
   6 |     """
   7 |     Calculates a deterministic heuristic representing scene congestion.
   8 |     
   9 |     Formula combines:
  10 |     - Base weight per object detected.
  11 |     - Total percentage of image pixel area covered by bounding boxes.
  12 |     - (Future/Optional) Depth distribution.
  13 |     
  14 |     Returns a score typically between 0.0 (empty) and 1.0 (highly congested).
  15 |     """
  16 |     total_area = float(image_width * image_height)
  17 |     if total_area == 0:
  18 |         return 0.0
  19 |         
  20 |     object_count = len(spatial_result.object_distances)
  21 |     
  22 |     # 1. Base object weight (e.g., 0.05 per object)
  23 |     count_congestion = min(1.0, object_count * 0.05)
  24 |     
  25 |     # 2. Bounding box coverage
  26 |     covered_area = 0.0
  27 |     for obj in spatial_result.object_distances:
  28 |         covered_area += calculate_bbox_area(obj.detected_object.bbox)
  29 |         
  30 |     coverage_ratio = min(1.0, covered_area / total_area)
  31 |     
  32 |     # Simple weighted average
  33 |     congestion_index = (count_congestion * 0.4) + (coverage_ratio * 0.6)
  34 |     
  35 |     return min(1.0, congestion_index)
  36 | 
  37 | 
  38 | def evaluate_placement_region(spatial_result: SpatialAnalysisResult) -> Tuple[bool, float]:
  39 |     """
  40 |     Infers whether the Available Placement Region is reasonably clear.
  41 |     
  42 |     Uses deterministic constraints:
  43 |     - Relies on nearest object inverse depth (high value = close).
  44 |     - Ensures the nearest object is not overwhelmingly close.
  45 |     
  46 |     Returns:
  47 |         (is_region_clear, nearest_obstacle_depth)
  48 |     """
  49 |     if not spatial_result.nearest_object:
  50 |         # No objects detected -> region is clear
  51 |         return True, 0.0
  52 |         
  53 |     nearest_depth = spatial_result.nearest_object.estimated_depth
  54 |     
  55 |     # Heuristic: Since MiDaS outputs are inverse depths, higher = closer.
  56 |     # We compare the nearest depth to the average scene depth.
  57 |     # If the nearest object is significantly closer than the mean, we flag the region as constrained.
  58 |     # Note: These values are relative, not absolute metric units.
  59 |     
  60 |     mean_depth = spatial_result.analysis_metadata.get("mean_depth", 0.0)
  61 |     
  62 |     # If the closest object is more than 2x the mean inverse depth, it's very close to the camera.
  63 |     if mean_depth > 0 and nearest_depth > (mean_depth * 2.0):
  64 |         return False, nearest_depth
  65 |         
  66 |     return True, nearest_depth

  30 |             limiting_factor = "OBSTACLE_PROXIMITY"

  34 |     # to see if rotating the object makes a difference to its perceived normalized footprint constraint.

  10 | def associate_depth(detected_object: DetectedObject, depth_map: np.ndarray) -> ObjectDistance:
  11 |     """
  12 |     Associates a DetectedObject with a representative depth scalar from the depth map.
  13 |     
  14 |     Uses the median depth of the pixels inside the object's bounding box.
  15 |     Median is robust to outliers such as noise or minor bounding box inaccuracies.
  16 |     
  17 |     Args:
  18 |         detected_object (DetectedObject): The object to evaluate.
  19 |         depth_map (np.ndarray): The 2D depth map (H, W).
  20 |         
  21 |     Returns:
  22 |         ObjectDistance: The populated DTO.
  23 |     """
  24 |     bbox = detected_object.bbox
  25 |     
  26 |     # Constrain bounding box to image dimensions
  27 |     height, width = depth_map.shape
  28 |     x1 = max(0, int(bbox.x1))
  29 |     y1 = max(0, int(bbox.y1))
  30 |     x2 = min(width, int(bbox.x2))
  31 |     y2 = min(height, int(bbox.y2))
  32 |     
  33 |     # Extract depth values within the bounding box
  34 |     region = depth_map[y1:y2, x1:x2]
  35 |     
  36 |     if region.size == 0:
  37 |         # Fallback if bounding box is completely outside or invalid
  38 |         median_depth = 0.0
  39 |     else:
  40 |         median_depth = float(np.median(region))
  41 |         
  42 |     center = get_bbox_center(bbox)
  43 |     
  44 |     return ObjectDistance(
  45 |         detected_object=detected_object,
  46 |         estimated_depth=median_depth,
  47 |         bbox_center=center,
  48 |         confidence=1.0
  49 |     )
  50 | 
  51 | def sort_by_depth(objects: List[ObjectDistance]) -> List[ObjectDistance]:
  52 |     """
  53 |     Sorts the objects by depth.
  54 |     
  55 |     Note: In MiDaS, larger values indicate closer proximity (disparity).
  56 |     We sort descending so the nearest object is first.
  57 |     
  58 |     Args:
  59 |         objects (List[ObjectDistance]): The list to sort.
  60 |         
  61 |     Returns:
  62 |         List[ObjectDistance]: Sorted list (closest first).
  63 |     """
  64 |     return sorted(objects, key=lambda obj: obj.estimated_depth, reverse=True)
  65 | 
  66 | def calculate_scene_statistics(depth_map: np.ndarray, object_count: int) -> Dict[str, float]:
  67 |     """
  68 |     Calculates basic statistical properties of the scene based on the depth map.
  69 |     
  70 |     Args:
  71 |         depth_map (np.ndarray): The 2D depth map.

  75 |         Dict[str, float]: A dictionary containing statistical metrics.

  78 |         return {
  79 |             "object_count": float(object_count),

  81 |             "min_depth": 0.0,

   6 | class SpatialAnalysisEngine:

   8 |     Combines DetectionResult and DepthResult into a deterministic spatial foundation.
   9 |     
  10 |     This engine operates purely on geometric calculations and statistical reductions 
  11 |     over the provided DTOs. It contains no AI/ML inference logic.

  13 |     
  14 |     def analyze(self, detection: DetectionResult, depth: DepthResult) -> SpatialAnalysisResult:
  15 |         """
  16 |         Executes the spatial analysis pipeline to merge detection and depth data.
  17 |         
  18 |         Args:
  19 |             detection (DetectionResult): The result from the object detection model.
  20 |             depth (DepthResult): The result from the depth estimation model.
  21 |             
  22 |         Returns:
  23 |             SpatialAnalysisResult: The combined deterministic spatial understanding.
  24 |         """
  25 |         
  26 |         # 1. Depth Association for all detected objects
  27 |         object_distances = []
  28 |         for obj in detection.objects:
  29 |             obj_dist = associate_depth(obj, depth.depth_map)
  30 |             object_distances.append(obj_dist)
  31 |             
  32 |         # 2. Relative Ordering
  33 |         depth_order = sort_by_depth(object_distances)
  34 |         
  35 |         nearest_object = depth_order[0] if depth_order else None
  36 |         furthest_object = depth_order[-1] if depth_order else None
  37 |         
  38 |         # 3. Scene Statistics
  39 |         analysis_metadata = calculate_scene_statistics(
  40 |             depth_map=depth.depth_map, 
  41 |             object_count=len(detection.objects)
  42 |         )
  43 |         
  44 |         return SpatialAnalysisResult(
  45 |             object_distances=object_distances,

  35 | def calculate_pixel_distance(p1: Tuple[float, float], p2: Tuple[float, float]) -> float:
  36 |     """
  37 |     Calculates the Euclidean distance between two points in pixel space.
  38 |     
  39 |     Args:
  40 |         p1 (Tuple[float, float]): The first point (x, y).
  41 |         p2 (Tuple[float, float]): The second point (x, y).
  42 |         
  43 |     Returns:
  44 |         float: The Euclidean distance.
  45 |     """
  46 |     return math.hypot(p2[0] - p1[0], p2[1] - p1[1])

  48 | def calculate_overlap_area(bbox1: BoundingBox, bbox2: BoundingBox) -> float:
  49 |     """
  50 |     Calculates the intersection area of two bounding boxes.
  51 |     
  52 |     Args:
  53 |         bbox1 (BoundingBox): The first bounding box.
  54 |         bbox2 (BoundingBox): The second bounding box.
  55 |         
  56 |     Returns:
  57 |         float: The overlapping area in square pixels.
  58 |                Returns 0.0 if there is no overlap.
  59 |     """
  60 |     inter_x1 = max(bbox1.x1, bbox2.x1)
  61 |     inter_y1 = max(bbox1.y1, bbox2.y1)
  62 |     inter_x2 = min(bbox1.x2, bbox2.x2)
  63 |     inter_y2 = min(bbox1.y2, bbox2.y2)
  64 |     
  65 |     inter_width = max(0.0, inter_x2 - inter_x1)
  66 |     inter_height = max(0.0, inter_y2 - inter_y1)
  67 |     
  68 |     return inter_width * inter_height

  17 |     def recommend(self, request: RecommendationRequest) -> RecommendationResponse:
  18 |         start_time = time.perf_counter()
  19 |         
  20 |         prefs = request.preferences
  21 |         products = request.available_products
  22 |         
  23 |         total_evaluated = len(products)
  24 |         
  25 |         # Helper for tokenization
  26 |         def _tokenize(text: str) -> Set[str]:
  27 |             if not text:
  28 |                 return set()
  29 |             # Normalize to lowercase and extract alphabetic tokens
  30 |             tokens = re.findall(r'[a-z]+', text.lower())
  31 |             return set(tokens)
  32 |             
  33 |         query_tokens = _tokenize(prefs.query) if prefs.query else set()
  34 |         req_category = prefs.category.lower().strip() if prefs.category else None
  35 |         req_material = prefs.material.lower().strip() if prefs.material else None
  36 |         req_max_price = prefs.max_price
  37 |         
  38 |         eligible_products: List[Tuple[Any, int, List[str]]] = []
  39 |         
  40 |         for prod in products:
  41 |             # Hard Filters
  42 |             if prod.stock <= 0:
  43 |                 continue
  44 |             if req_max_price is not None and prod.price > req_max_price:
  45 |                 continue
  46 |             if req_category is not None and prod.category.lower().strip() != req_category:
  47 |                 continue
  48 |                 
  49 |             # Soft Scoring
  50 |             score = 0
  51 |             reasons = []
  52 |             
  53 |             # Material Match
  54 |             if req_material and prod.material:
  55 |                 if req_material in prod.material.lower():
  56 |                     score += 50
  57 |                     reasons.append(f"Material match: {prod.material}")
  58 |                     
  59 |             # Query Keyword Match
  60 |             if query_tokens:
  61 |                 prod_text = f"{prod.name} {prod.description}"
  62 |                 prod_tokens = _tokenize(prod_text)
  63 |                 
  64 |                 matched_tokens = query_tokens.intersection(prod_tokens)
  65 |                 if matched_tokens:
  66 |                     points = len(matched_tokens) * 10
  67 |                     score += points
  68 |                     reasons.append(f"Keyword match ({len(matched_tokens)} tokens, +{points} pts)")
  69 |                     
  70 |             if score == 0:
  71 |                 reasons.append("Eligible product (Score: 0)")
  72 |                 
  73 |             eligible_products.append((prod, score, reasons))
  74 |             
  75 |         # Ranking: Descending score, then tie-break by productId alphabetically
  76 |         eligible_products.sort(key=lambda x: (-x[1], x[0].productId))
  77 |         
  78 |         recommended_ids = []
  79 |         matching_info = {}
  80 |         
  81 |         for prod, score, reasons in eligible_products:
  82 |             recommended_ids.append(prod.productId)
  83 |             # Create a localized dict payload for the DTO
  84 |             matching_info[prod.productId] = {
  85 |                 "score": score,
  86 |                 "reasons": reasons
  87 |             }
  88 |             
  89 |         end_time = time.perf_counter()
